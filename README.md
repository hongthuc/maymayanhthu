    ];

    // Hàm tạo SVG dự phòng nếu mạng chặn ảnh Google Drive
    function createFallbackSvgDataUri(label) {
      const svg = `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 300 220" width="300" height="220">
        <rect width="300" height="220" fill="#F8FAFC"/>
        <rect x="45" y="155" width="210" height="18" rx="6" fill="#0B4F9E"/>
        <path d="M75 55 H215 C228 55 235 63 235 75 V155 H175 V95 H115 V130 H75 Z" fill="#1E40AF"/>
        <rect x="88" y="130" width="12" height="24" fill="#E11D2E"/>
        <circle cx="205" cy="80" r="12" fill="#E11D2E"/>
        <text x="150" y="198" text-anchor="middle" font-family="Inter, sans-serif" font-size="13" font-weight="bold" fill="#0B4F9E">${label}</text>
      </svg>`;
      return 'data:image/svg+xml;utf8,' + encodeURIComponent(svg);
    }

    // Xử lý tự động chuyển đổi nguồn ảnh Google Drive nếu endpoint đầu tiên bị lỗi
    function attachDriveImageFallbacks() {
      const imgs = document.querySelectorAll('img.drive-img');
      imgs.forEach((img) => {
        if (img.dataset.boundFallback) return;
        img.dataset.boundFallback = 'true';
        img.addEventListener('error', function () {
          const id = this.getAttribute('data-drive-id');
          const step = parseInt(this.dataset.fallbackStep || '0', 10);
          if (step === 0 && id) {
            this.dataset.fallbackStep = '1';
            this.src = `https://drive.google.com/thumbnail?id=${id}&sz=w800`;
          } else if (step === 1 && id) {
            this.dataset.fallbackStep = '2';
            this.src = `https://drive.google.com/uc?export=view&id=${id}`;
          } else {
            this.dataset.fallbackStep = '3';
            this.src = createFallbackSvgDataUri('MÁY MAY ANH THƯ');
          }
        });
      });
    }

    // Hàm hiển thị các danh mục sản phẩm ra trang chủ
    // Chia 2 cột trên điện thoại (grid-cols-2), 4 cột trên máy tính (lg:grid-cols-4)
    function renderProductSections(filterCode = 'ALL', searchQuery = '') {
      const container = document.getElementById('productSectionsContainer');
      const normalizedQuery = searchQuery.trim().toLowerCase();

      let sectionsToRender = PRODUCT_SECTIONS;
      if (filterCode !== 'ALL') {
        if (filterCode === 'BRUCE') {
          sectionsToRender = PRODUCT_SECTIONS.filter(s => s.code === 'OKITO');
        } else if (filterCode === 'PHU_TUNG') {
          sectionsToRender = PRODUCT_SECTIONS.filter(s => s.code === 'MAY_CU');
        } else {
          sectionsToRender = PRODUCT_SECTIONS.filter(s => s.code === filterCode);
        }
      }

      let html = '';
      sectionsToRender.forEach((section) => {
        const filteredProducts = section.products.filter(p =>
          !normalizedQuery || p.name.toLowerCase().includes(normalizedQuery)
        );
        if (filteredProducts.length === 0) return;

        html += `
          <section class="flatsome-container">
            <div class="flatsome-section-title rounded-t">
              <div class="flatsome-section-title-main rounded-tl">
                <span>${section.title}</span>
              </div>
              <button type="button" onclick="filterByCategory('${section.code}')" class="text-xs font-semibold text-brand hover:text-brand-accent pr-3 transition shrink-0">
                Xem thêm <i class="fa-solid fa-angle-right ml-0.5"></i>
              </button>
            </div>

            <!-- Lưới sản phẩm: 2 cột trên điện thoại (grid-cols-2), 4 cột trên máy tính (lg:grid-cols-4) -->
            <div class="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-4 gap-2.5 sm:gap-4">
              ${filteredProducts.map((prod) => `
                <div class="bg-white rounded border border-slate-200 hover:border-brand shadow-2xs hover:shadow-md transition flex flex-col justify-between overflow-hidden relative group">
                  <!-- Nhãn % giảm giá góc trên bên trái giống Flatsome -->
                  <div class="flatsome-badge-sale">${prod.discount}</div>

                  <div>
                    <!-- Ảnh sản phẩm thực tế từ Google Drive -->
                    <div class="w-full h-36 sm:h-48 bg-white p-2.5 flex items-center justify-center border-b border-slate-100 overflow-hidden">
                      <img
                        src="https://lh3.googleusercontent.com/d/${prod.driveId}=w600"
                        data-drive-id="${prod.driveId}"
                        alt="${prod.name}"
                        class="drive-img max-h-full max-w-full object-contain group-hover:scale-105 transition duration-300"
                        loading="lazy"
                      >
                    </div>

                    <!-- Tên sản phẩm đầy đủ, không bị cắt chữ hoặc tràn ra ngoài -->
                    <div class="p-2.5 sm:p-3 pb-1.5">
                      <div class="text-[10px] font-semibold text-slate-400 uppercase mb-0.5">${section.title}</div>
                      <h3 class="product-title-full text-xs sm:text-[13px] font-bold text-slate-800 group-hover:text-brand">
                        ${prod.name}
                      </h3>
                    </div>
                  </div>

                  <!-- Giá giảm giá chuẩn Flatsome + Nút nhận báo giá -->
                  <div class="px-2.5 sm:px-3 pb-2.5 sm:pb-3 pt-1">
                    <div class="flex flex-wrap items-baseline gap-x-2 gap-y-0.5 mb-2">
                      <del class="text-[11px] text-slate-400">${prod.oldPrice}</del>
                      <ins class="text-xs sm:text-sm font-extrabold text-brand-accent no-underline">${prod.salePrice}</ins>
                    </div>
                    <div class="grid grid-cols-2 gap-1.5">
                      <a href="#bao-gia" onclick="selectProductForQuote('${prod.name.replace(/'/g, "\\'")}')" class="text-center bg-brand hover:bg-brand-dark text-white text-[11px] font-semibold py-1.5 px-2 rounded transition">
                        Báo giá
                      </a>
                      <a href="tel:0935747727" class="text-center bg-slate-100 hover:bg-brand-light text-brand border border-brand/20 text-[11px] font-semibold py-1.5 px-2 rounded transition">
                        Liên hệ
                      </a>
                    </div>
                  </div>
                </div>
              `).join('')}
            </div>
          </section>
        `;
      });

      if (!html) {
        html = `
          <div class="flatsome-container">
            <div class="bg-white rounded border border-slate-200 p-6 text-center space-y-2">
              <div class="text-sm font-bold text-slate-700">Không tìm thấy sản phẩm phù hợp với từ khóa "${searchQuery}"</div>
              <button type="button" onclick="filterByCategory('ALL')" class="bg-brand text-white text-xs font-bold px-4 py-2 rounded">
                Hiển thị tất cả sản phẩm
              </button>
            </div>
          </div>
        `;
      }

      container.innerHTML = html;
      attachDriveImageFallbacks();
    }

    // Lọc theo danh mục
    function filterByCategory(code) {
      renderProductSections(code, '');
      const el = document.getElementById('san-pham-all');
      if (el && code !== 'ALL') {
        el.scrollIntoView({ behavior: 'smooth', block: 'start' });
      }
    }

    // Điền nhanh tên máy vào Form báo giá khi khách bấm "Báo giá" trên sản phẩm
    function selectProductForQuote(productName) {
      const input = document.getElementById('quoteProduct');
      if (input) {
        input.value = productName;
      }
    }

    // Điều khiển Slider 2 ảnh Banner tự trượt
    let currentBannerIndex = 0;
    function setBannerSlide(index) {
      currentBannerIndex = index;
      const desktopSlides = document.querySelectorAll('.banner-slide-desktop');
      const mobileSlides = document.querySelectorAll('.banner-slide-mobile');
      const dot0 = document.getElementById('bannerDot0');
      const dot1 = document.getElementById('bannerDot1');

      desktopSlides.forEach((slide, i) => {
        if (i === index) {
          slide.classList.remove('opacity-0', 'pointer-events-none');
          slide.classList.add('opacity-100');
        } else {
          slide.classList.remove('opacity-100');
          slide.classList.add('opacity-0', 'pointer-events-none');
        }
      });

      mobileSlides.forEach((slide, i) => {
        if (i === index) {
          slide.classList.remove('opacity-0', 'pointer-events-none');
          slide.classList.add('opacity-100');
        } else {
          slide.classList.remove('opacity-100');
          slide.classList.add('opacity-0', 'pointer-events-none');
        }
      });

      if (dot0 && dot1) {
        if (index === 0) {
          dot0.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
          dot1.className = 'w-2 h-1.5 rounded-full bg-white/60 transition-all';
        } else {
          dot0.className = 'w-2 h-1.5 rounded-full bg-white/60 transition-all';
          dot1.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
        }
      }
    }

        });
      }
